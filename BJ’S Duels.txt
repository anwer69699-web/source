local function getNearestOtherPlayer()
    local character = player.Character
    local myRoot = character and character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local closest, closestDist = nil, math.huge
    for _, plr in Players:GetPlayers() do
        if plr == player or not plr.Character then continue end
        local theirRoot = plr.Character:FindFirstChild("HumanoidRootPart")
        if not theirRoot then continue end
        local dist = (myRoot.Position - theirRoot.Position).Magnitude
        if dist < closestDist then
            closestDist = dist
            closest = plr
        end
    end
    return closest
end

local function cacheCharacterData()
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return false end
    cachedCharData = {character = char, humanoid = hum, root = root}
    return true
end

local function disconnectAll()
    for _, conn in ipairs(ragdollConnections) do pcall(function() conn:Disconnect() end) end
    ragdollConnections = {}
end

local function isRagdolled()
    if not cachedCharData.humanoid then return false end
    local state = cachedCharData.humanoid:GetState()
    local ragStates = {[Enum.HumanoidStateType.Physics] = true, [Enum.HumanoidStateType.Ragdoll] = true, [Enum.HumanoidStateType.FallingDown] = true}
    if ragStates[state] then return true end
    local endTime = player:GetAttribute("RagdollEndTime")
    return endTime and (endTime - workspace:GetServerTimeNow()) > 0
end

local function forceExitRagdoll()
    if not cachedCharData.humanoid or not cachedCharData.root then return end
    pcall(function() player:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow()) end)
    for _, d in ipairs(cachedCharData.character:GetDescendants()) do
        if d:IsA("BallSocketConstraint") or (d:IsA("Attachment") and d.Name:find("RagdollAttachment")) then d:Destroy() end
    end
    if not isBoosting then isBoosting = true cachedCharData.humanoid.WalkSpeed = BOOST_SPEED end
    if cachedCharData.humanoid.Health > 0 then cachedCharData.humanoid:ChangeState(Enum.HumanoidStateType.Running) end
end

local function v1HeartbeatLoop()
    while antiRagdollMode == "v1" do
        task.wait()
        local rag = isRagdolled()
        if rag then forceExitRagdoll()
        elseif isBoosting and not rag then
            isBoosting = false
            if cachedCharData.humanoid then cachedCharData.humanoid.WalkSpeed = DEFAULT_SPEED end
        end
    end
end

local function EnableAntiRagdoll()
    if antiRagdollMode == "v1" then return end
    if not cacheCharacterData() then return end
    antiRagdollMode = "v1"
    table.insert(ragdollConnections, RunService.RenderStepped:Connect(function()
        if workspace.CurrentCamera and cachedCharData.humanoid then workspace.CurrentCamera.CameraSubject = cachedCharData.humanoid end
    end))
    table.insert(ragdollConnections, player.CharacterAdded:Connect(function()
        isBoosting = false task.wait(0.5) cacheCharacterData()
    end))
    task.spawn(v1HeartbeatLoop)
end

local function DisableAntiRagdoll()
    antiRagdollMode = nil
    if isBoosting and cachedCharData.humanoid then cachedCharData.humanoid.WalkSpeed = DEFAULT_SPEED end
    isBoosting = false disconnectAll() cachedCharData = {}
end

local IsStealing, StealProgress, CurrentTargetUID, allAnimalsCache = false, 0, nil, {}

local function scanPlots()
    if not autoGrabEnabled then return end
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return end
    local newCache = {}
    for _, plot in ipairs(plots:GetChildren()) do
        local sign = plot:FindFirstChild("PlotSign")
        local isMine = sign and sign:FindFirstChild("YourBase") and sign.YourBase.Enabled
        if not isMine then
            local podiums = plot:FindFirstChild("AnimalPodiums")
            if podiums then
                for _, podium in ipairs(podiums:GetChildren()) do
                    local base = podium:FindFirstChild("Base")
                    if base then
                        table.insert(newCache, {podium = podium, worldPosition = podium:GetPivot().Position, uid = plot.Name.."_"..podium.Name})
                    end
                end
            end
        end
    end
    allAnimalsCache = newCache
end

local function executeSteal(prompt, uid, pos)
    if IsStealing or not autoGrabEnabled then return end
    IsStealing = true
    CurrentTargetUID = uid
    task.spawn(function()
        local duration, start = 0.55, os.clock()
        while autoGrabEnabled and CurrentTargetUID == uid do
            local elapsed = os.clock() - start
            local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not hrp or (hrp.Position - pos).Magnitude > (AUTO_STEAL_PROX_RADIUS + 5) then break end
            StealProgress = math.clamp(elapsed/duration, 0, 1)
            if StealProgress >= 1 then fireproximityprompt(prompt) break end
            RunService.Heartbeat:Wait()
        end
        IsStealing, StealProgress, CurrentTargetUID = false, 0, nil
    end)
end

local function setVisual(btn, active)
    TweenService:Create(btn, TweenInfo.new(0.2), {TextColor3 = active and ON_GREEN or TEXT_WHITE}):Play()
end

local function setupButton(btn, callback)
    local dragging, startPos, dragStart, moved = false, nil, nil, false
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos, moved = true, input.Position, btn.Position, false
        end
    end)
    btn.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > DRAG_THRESHOLD then moved = true end
            btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            if not moved then callback() end
        end
    end)
end

RunService.Heartbeat:Connect(function()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChild("Humanoid")
    if not hrp or not hum then return end
    
    if speedBoostEnabled and hum.MoveDirection.Magnitude > 0 then
        local dir = hum.MoveDirection
        hrp.AssemblyLinearVelocity = Vector3.new(dir.X * _G.SpeedValue, hrp.AssemblyLinearVelocity.Y, dir.Z * _G.SpeedValue)
    end

    if batAimEnabled then
        if not currentTarget or not currentTarget.Parent or not currentTarget.Character then
            currentTarget = getNearestOtherPlayer()
        end
        if currentTarget and currentTarget.Character then
            local tRoot = currentTarget.Character:FindFirstChild("HumanoidRootPart")
            if tRoot then
                local desiredPos = tRoot.Position + (-tRoot.CFrame.LookVector * OFFSET_DISTANCE)
                local diff = desiredPos - hrp.Position
                if diff.Magnitude > STOP_THRESHOLD then
                    hrp.AssemblyLinearVelocity = diff.Unit * math.min(_G.BatAimSpeed, diff.Magnitude * CATCHUP_BOOST_MULT)
                else
                    hrp.AssemblyLinearVelocity = tRoot.AssemblyLinearVelocity
                end
            end
        end
        local bat = player.Backpack:FindFirstChild("Bat") or char:FindFirstChild("Bat")
        if bat then
            if bat.Parent == player.Backpack then hum:EquipTool(bat) end
            bat:Activate()
        end
    end

    if patrolMode ~= "none" then
        local waypoints = (patrolMode == "right" and rightWaypoints or leftWaypoints)
        local target = waypoints[currentWaypoint]
        local dist = (Vector3.new(target.X, 0, target.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Magnitude
        if dist > 3 then
            local speed = (currentWaypoint >= 3 and 29.4 or 60)
            local dir = (Vector3.new(target.X, 0, target.Z) - Vector3.new(hrp.Position.X, 0, hrp.Position.Z)).Unit
            hrp.AssemblyLinearVelocity = Vector3.new(dir.X * speed, hrp.AssemblyLinearVelocity.Y, dir.Z * speed)
        else
            currentWaypoint = (currentWaypoint >= #waypoints) and 1 or (currentWaypoint + 1)
        end
    end

    if autoGrabEnabled then
        Fill.Size = UDim2.new(math.clamp(StealProgress, 0, 1), 0, 1, 0)
        if not IsStealing then
            for _, animal in ipairs(allAnimalsCache) do
                if (hrp.Position - animal.worldPosition).Magnitude <= _G.GrabRadius then
                    local prompt = animal.podium:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt then executeSteal(prompt, animal.uid, animal.worldPosition) break end
                end
            end
        end
    end

    if spinEnabled then
        if not hrp:FindFirstChild("SpinVelocity") then
            local av = Instance.new("BodyAngularVelocity", hrp)
            av.Name = "SpinVelocity"
            av.MaxTorque = Vector3.new(0, math.huge, 0)
        end
        hrp.SpinVelocity.AngularVelocity = Vector3.new(0, _G.SpinSpeed, 0)
    elseif hrp:FindFirstChild("SpinVelocity") then
        hrp.SpinVelocity:Destroy()
    end
end)
