local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local Player = Players.LocalPlayer
local Character = Player.Character or Player.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

-- ============================================================
--                    CONFIGURATION
-- ============================================================

local CONFIG = {
    UPDATE_INTERVAL = 0.1,  -- تحديث كل 0.1 ثانية
    DECIMAL_PLACES = 2,     -- عدد المنازل العشرية
    SHOW_NOTIFICATIONS = true,
    AUTO_COPY = true        -- نسخ تلقائي للحافظة
}

-- ============================================================
--                    UTILITY FUNCTIONS
-- ============================================================

-- دالة لنسخ النص للحافظة
local function CopyToClipboard(text)
    local success = pcall(function()
        if setclipboard then
            setclipboard(text)
            return true
        elseif toclipboard then
            toclipboard(text)
            return true
        elseif syn and syn.write_clipboard then
            syn.write_clipboard(text)
            return true
        end
        return false
    end)
    return success
end

-- دالة لتنسيق الإحداثيات - فقط الأرقام بالتنسيق: -485.17, -5.15, 25.08
local function FormatCoordinates(position)
    local format_str = "%." .. CONFIG.DECIMAL_PLACES .. "f"
    return string.format(format_str .. ", " .. format_str .. ", " .. format_str,
        position.X, position.Y, position.Z)
end

-- دالة لإظهار الإشعارات
local function Notify(title, text, duration)
    duration = duration or 3
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = duration
        })
    end)
end

-- ============================================================
--                    MAIN LOGIC
-- ============================================================

local lastCoords = ""
local isRunning = true

-- تحديث الإحداثيات
local function UpdateCoordinates()
    if not isRunning then return end
    if not HumanoidRootPart then return end

    local position = HumanoidRootPart.Position
    local coordsText = FormatCoordinates(position)

    CoordsLabel.Text = coordsText

    -- نسخ تلقائي إذا تغيرت الإحداثيات
    if CONFIG.AUTO_COPY and coordsText ~= lastCoords then
        CopyToClipboard(coordsText)
        lastCoords = coordsText
    end
end

-- نسخ يدوي
CopyButton.MouseButton1Click:Connect(function()
    local position = HumanoidRootPart and HumanoidRootPart.Position
    if position then
        local coordsText = FormatCoordinates(position)
        local success = CopyToClipboard(coordsText)

        if success then
            CopyButton.Text = "COPIED!"
            CopyButton.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
            if CONFIG.SHOW_NOTIFICATIONS then
                Notify("Success!", "Coordinates copied: " .. coordsText, 2)
            end
        else
            CopyButton.Text = "FAILED"
            CopyButton.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
        end

        task.delay(1, function()
            CopyButton.Text = "COPY NOW"
            CopyButton.BackgroundColor3 = Color3.fromRGB(0, 200, 120)
        end)
    end
end)

-- إغلاق
CloseButton.MouseButton1Click:Connect(function()
    isRunning = false
    ScreenGui:Destroy()
    print("[Coords Tracker] Stopped")
end)

-- Loop للتحديث
RunService.Heartbeat:Connect(function()
    if isRunning then
        UpdateCoordinates()
    end
end)

-- إعادة التعيين عند respawn
Player.CharacterAdded:Connect(function(newChar)
    Character = newChar
    HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")
end)

-- إشعار البدء

    Notify("Coords Tracker", "Format: X, Y, Z | Auto-copy ON", 3)
end
