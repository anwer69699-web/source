-- Toggle panel open/closed
local panelVisible = true

local function setPanel(visible)
    panelVisible    = visible
    MainFrame.Visible = visible
    OpenButton.Visible = not visible
end

CloseButton.MouseButton1Click:Connect(function()
    setPanel(false)
end)

OpenButton.MouseButton1Click:Connect(function()
    setPanel(true)
end)

-- U key toggles panel
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.U then
        setPanel(not panelVisible)
    end
end)

-- Jump request pass-through for Inf Jump
UserInputService.JumpRequest:Connect(function()
    if getInfJump() then
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
            end
        end)
    end
end)

-- Drag support for main panel
do
    local dragging, dragInput, dragStart, startPos

    HeaderFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
           or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = MainFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                      or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            MainFrame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- HUD Stroke gradient spin + FPS/Ping updater
local gradientAngle = 0
local lastFrameTime = tick()
local frameCount    = 0
local fps           = 0

RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    local now  = tick()
    if now - lastFrameTime >= 1 then
        fps           = math.floor(frameCount / (now - lastFrameTime))
        frameCount    = 0
        lastFrameTime = now
    end

    -- Spinning stroke gradient
    gradientAngle = (gradientAngle + 2) % 360
    pcall(function() HudStrokeGradient.Rotation = gradientAngle end)

    -- Ping via Stats
    local ping = 0
    pcall(function()
        ping = math.floor(Stats.Network.ServerStatsItem["DataPing"]:GetValue())
    end)

    FpsPingLabel.Text = "FPS: " .. tostring(fps) .. " | Ping: " .. tostring(ping) .. "ms"
end)

-- Heartbeat: anti-ragdoll, inf health, walk speed
RunService.Heartbeat:Connect(function()
    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end

        -- Godmode keepalive
        if hum.MaxHealth ~= math.huge then
            hum.MaxHealth = math.huge
            hum.Health    = math.huge
        end

        -- Anti-ragdoll
        if getAntiRagdoll() then
            hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
            hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        end
    end)
end)

-- Auto-save every 10 seconds
task.spawn(function()
    while task.wait(10) do
        pcall(function() SaveButton.MouseButton1Click:Fire() end)
    end
end)

-- Fade-in animation on load
task.spawn(function()
    for i = 1, 10 do
        local t = i / 10
        OpenButton.BackgroundTransparency = 0.9 - (0.9 * t)
        OpenButton.TextTransparency       = 1 - t
        task.wait(0.05)
    end
end)
